Cluster-Aware Producer

pom.xml <dependency>
  <groupId>org.springframework.kafka</groupId>
  <artifactId>spring-kafka</artifactId>
</dependency>
@Configuration
public class ClusterProducerConfig {
  @Bean
  public ProducerFactory<String, String> producerFactory() {
    Map<String, Object> config = new HashMap<>();
    config.put(ProducerConfig.BOOTSTRAP_SERVERS_CONFIG, "localhost:9092,localhost:9093,localhost:9094");
    config.put(ProducerConfig.ACKS_CONFIG, "all"); // Wait for all replicas
    config.put(ProducerConfig.RETRIES_CONFIG, 10); // Automatic retry on failure
    return new DefaultKafkaProducerFactory<>(config);
  }
}
