public Map<String, Object> getClusterHealth() throws Exception {
  DescribeClusterResult cluster = adminClient.describeCluster();
  int brokers = cluster.nodes().get().size();
  int topics = adminClient.listTopics().names().get().size();
  Map<String, Object> health = new HashMap<>();
  health.put("brokers", brokers);
  health.put("topics", topics);
  health.put("status", brokers >= 2 ? "healthy" : "degraded");
  return health;
}
