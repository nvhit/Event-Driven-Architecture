Phát triển API nâng cao

  • /cluster/health → Trạng thái tổng thể của cluster
  • /cluster/metadata → Cấu trúc mạng của Broker
  • /consumers/stats → Số liệu Consumer group
  • /cluster/simulate-failure → Kiểm thử khả năng chịu lỗi
@Component
public class ConsumerBootstrap {
  @EventListener(ApplicationReadyEvent.class)
  public void startConsumers() {
    // Start multiple consumers
    for (int i = 0; i < 3; i++) {
      var consumer = new ClusterEventConsumer("consumer-" + i);
      consumer.startConsuming();
    }
  }
}
